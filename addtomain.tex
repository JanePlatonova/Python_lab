% addtomain.tex
% Things from pylab.sty that main.tex does not have.
% Usage: in main.tex put  \input{addtomain}  in the preamble,
%        AFTER \definecolor{pyblue}/{pyyellow} and \usepackage{fontawesome}
%        (i.e. just before \begin{document}).
%
% Left out on purpose (they conflict with the ONU format in main.tex):
%   kpfonts, mathastext, Roboto fonts, geometry 1.75cm, \linespread{1.1},
%   \parindent 15pt, \pagestyle{empty}, babel[english,russian], hypersetup.

% ---------- Packages ----------
\usepackage{wrapfig}     % figures with text wrapped around them
\usepackage{textcomp}    % extra text symbols
\usepackage{mathrsfs}    % \mathscr
\usepackage{array}       % needed for \newcolumntype

% ---------- TikZ libraries ----------
\usetikzlibrary{arrows}
\usetikzlibrary{arrows.meta}
\usetikzlibrary{shadows}
\usetikzlibrary{shadows.blur}
\usetikzlibrary{shapes}

% ---------- Tables ----------
\renewcommand{\arraystretch}{1.1}

% Format a whole table row: use $ for the first column and ^ for the others, then \rowstyle{\bfseries}
\newcolumntype{$}{>{\global\let\currentrowstyle\relax}}
\newcolumntype{^}{>{\currentrowstyle}}
\newcommand{\rowstyle}[1]{\gdef\currentrowstyle{#1}%
	#1\ignorespaces
}

% ---------- Commands ----------
% Lab header with the Python logo (needs Py.png)
\newcommand{\myheader}[1]{
	\begin{tcolorbox}[enhanced, fonttitle=\scriptsize\sffamily\bfseries, coltitle=pyyellow, colframe=white, title style={left color=pyblue, right color=black}, colback=white, boxrule=0pt, leftrule=0pt, rightrule=0pt, titlerule=0pt, toprule=0pt, bottomrule=1pt, arc=0pt, outer arc=0pt, left=0mm, right=0mm, adjusted title={Курс <<Основи програмування>>\hbox to 1.75cm {}\\Спеціальність <<Математика>>\hbox to 1.75cm {} }, halign title=flush right, boxsep=0mm, toptitle=1.5mm, bottomtitle=1.5mm]
		\begin{minipage}[b]{0.75\textwidth}
			{\Large\sffamily\bfseries #1}
			\vspace{-0.3cm}
		\end{minipage}
		\hfill
		\begin{minipage}{0.15\textwidth}
			\vspace{-0.65cm}
			\includegraphics[width=\linewidth]{Py.png}
			\vspace{-0.6cm}
		\end{minipage}
		\par\vspace{6pt}
		\begin{tikzpicture}
			\fill[left color=pyblue, right color=black] (0,0) rectangle (\textwidth,-1pt);
		\end{tikzpicture}
	\end{tcolorbox}
	\bigskip
}

% Variant heading (Варіант = Ukrainian; pylab.sty had Russian "Вариант")
\newcommand{\variant}[1]{
	\noindent\textsf{\large\bfseries Варіант~#1}
}

% Link that opens in a new window, with an icon
\newcommand{\myurl}[2]{
	\href[pdfnewwindow=true]{#1}{#2~\faExternalLink}
}

% Inline Python code.
% NOTE: pylab.sty calls this \py, but main.tex already has an environment
% named "py" (\newtcblisting{py}), which defines \py. So we call it \pyi.
% main.tex's \mi does the same thing.
\newcommand{\pyi}[1]{\mintinline{python}{#1}}

% Number or letter in a filled circle
\newcommand*\circled[1]{\tikz[baseline=(char.base)]{
		\node[shape=circle,draw=pyblue, fill=pyblue,inner sep=2pt] (char) {\color{pyyellow}\bfseries\sffamily#1};}}

% ---------- Lists ----------
% enumerate items get circled letters: (A), (B), ...
\setlist[enumerate]{
	label=\protect\circled{\Alph*},
	itemsep=6pt
}
