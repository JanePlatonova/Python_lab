% Створення команди \LabHeader{Лабораторна}{Варіант}
\newcommand{\LabHeader}[2]{%
	\noindent
	\begin{tikzpicture}[remember picture]
		% 1. Градієнтна смуга зверху
		\shade[left color=lablightblue, right color=black!90] 
		(0,0) rectangle (\textwidth, 0.7cm);
		
		% 2. Текст "Лабораторна X" білим кольором справа на смузі
		\node[anchor=east, pyyellow, font=\bfseries\sffamily\large] 
		at (\textwidth - 1.5cm, 0.35cm) {Лабораторна #1};
		
		% 3. Текст "Варіант Y" знизу ліворуч
		\node[anchor=west, black, font=\bfseries\Large] 
		at (0, -0.6cm) {Варіант #2};
		
		% 4. Зображення Python (з виходом за межі смуги)
		% Переконайтеся, що файл Py.png існує в папці проєкту
		\node[anchor=north east] 
		at (\textwidth + 0.2cm, 0.8cm) {\includegraphics[width=1.8cm]{Py.png}};
		
		% 5. Нижня роздільна лінія
		\draw[line width=0.8pt, lablightblue] 
		(0, -1.1cm) -- (\textwidth, -1.1cm);
	\end{tikzpicture}%
	\vspace{0.5cm} % Відступ після шапки
}
